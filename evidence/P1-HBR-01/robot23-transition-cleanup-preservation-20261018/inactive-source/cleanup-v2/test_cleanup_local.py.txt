def result(action, host, proof, packet, start=NOW-20, finish=NOW-19):
    return {'status': 'CONFIRMED', 'binding': x.cleanup_binding(s), 'action': action, 'host': host,
            'boot_id': s['request']['execution']['boot_ids'][host], 'started_at': x.iso(start),
            'finished_at': x.iso(finish), 'input_sha256': x.sha(x.encoded(packet)), 'proof': proof}


def check_receipts():
    pairs = [{'path': path, 'present': True, 'kind': 'regular', 'device': 1, 'inode': i+1,
              'uid': 0, 'gid': 0, 'mode': 0o600, 'nlink': 1, 'bytes': 10, 'context': x.RUN_CONTEXT}
             for i, path in enumerate(x.pair_paths())]
    return {h: result('check', h, {'pair': copy.deepcopy(pairs), 'owner_verified': True,
                    'source_metadata': s['request']['execution']['source_metadata'] if h == x.CP else []}, {},
                    NOW-40, NOW-39) for h in x.HOSTS}


def packet():
    checks = check_receipts()
    proof = p.load(p.OP / 'proof-linkage/after/deliver/ACTUAL_PROOF.json')
    proof['observed_at'] = x.iso(NOW-20)

                         {'checks': checks}, NOW-30, NOW-19)
    return {'checks': checks, 'observation': observation, 'retired': []}


def append_retired(v, host, n):
    prior = copy.deepcopy(v)
    proof = {'state': 'RETIRED_CURRENT_PAIR', 'paths': x.pair_paths(),
             'checks_sha256': x.sha(x.encoded(v['checks'])),
             'observation_sha256': x.sha(x.encoded(v['observation'])),
             'previous_sha256': x.sha(x.encoded(v['retired']))}
    v['retired'].append(result('retire', host, proof, prior, NOW-18+n*2, NOW-17+n*2))

    def exercise_retire(self, failure=None, changed=False, forged_local=False, late_stop=False):
        v=packet();writes=[];effects=[]
        class J:
            def write(self,key,value):
                writes.append((key,copy.deepcopy(value)))
                if key=='result' and late_stop:raise x.Failure('HOST_DEADLINE')
        class Pair:
            def __init__(self,s,mods):effects.append('constructor')
            def retire_cleanup_pair(self):
                effects.append('retire_once')
                if failure:raise x.Failure(failure)
        RH=types.SimpleNamespace()
        oe=types.SimpleNamespace(deadline=lambda seconds:contextlib.nullcontext())
        lh=types.SimpleNamespace(write_new=lambda path,data:writes.append(('global_stop',json.loads(data))))
        mods=(c,RH,None,oe,None,None,lh,None)
        def read(path,*_):
            val=v['checks'][x.CP] if 'check' in str(path) else v['observation']
            if forged_local:val={}
            return x.encoded(val)
        pair=copy.deepcopy(v['checks'][x.CP]['proof']['pair'])
        if changed:pair[0]['inode']+=1
        with patch.object(x.socket,'gethostname',return_value=x.CP),patch.object(x.time,'time',return_value=NOW),\
             patch.object(x,'new_journal',return_value=J()),patch.object(x,'original_host_check'),\
             patch.object(x,'protected',side_effect=read),patch.object(x,'pair_metadata',return_value=pair),\
             patch.object(x,'CleanupPair',Pair):
            out=x.execute(s,'retire',v,mods,Path('/synthetic'),NOW,NOW+60)
        return out,writes,effects

    def exercise_pair_method(self, drift=False, expire_after_first=False):
        events=[]
        def identity(*args):events.append('identity')
        def owner(obj):
            self.assertIs(obj.request,s['request']);self.assertEqual(obj.plan,c.plan(s['request']))
            self.assertEqual(obj.inv,x.OLD);events.append('original_owner_method')
            return {'username':'SYNTHETIC','password':'SYNTHETIC_NEVER_REAL'}
        rh=types.SimpleNamespace(identity=identity,actor_absence=lambda *_:events.append('actor_check'),
             RebootHost=types.SimpleNamespace(credential=owner),absent=lambda _:True)
        lh=types.SimpleNamespace(fsync_parent=lambda path:events.append(('fsync',str(path))))
        reads={}
        def metadata(path):
            path=str(path).replace('\\','/');reads[path]=reads.get(path,0)+1
            inode=1+(1 if drift and reads[path]>1 else 0)
            return {'path':path,'inode':inode}
        mods=(c,rh,None,None,None,None,lh,metadata)
        clock=[NOW]
        def unlink(path):
            events.append(('unlink',path.as_posix()))
            if expire_after_first:clock[0]=x.timestamp(s['not_after'])
        with patch.object(x.socket,'gethostname',return_value=x.CP),\
             patch.object(x.time,'time',side_effect=lambda:clock[0]),patch.object(x,'checked_dir'),\
             patch.object(x,'protected',return_value=x.encoded(c.plan(s['request']))),\
             patch.object(x.Path,'unlink',autospec=True,side_effect=unlink):
            obj=x.CleanupPair(s,mods)
            if drift or expire_after_first:self.rejected(obj.retire_cleanup_pair)
            else:obj.retire_cleanup_pair()
        return events

    def test_same_owner_logic_and_exact_unlink_fsync_only(self):
        ev=self.exercise_pair_method()
        self.assertEqual(ev.count('original_owner_method'),1)
        self.assertEqual([v for v in ev if isinstance(v,tuple)],
                         [('unlink',x.pair_paths()[0]),('fsync',x.pair_paths()[0]),
                          ('unlink',x.pair_paths()[1]),('fsync',x.pair_paths()[1])])

    def collected_retire(self, late_stop=False, during_read=False, malformed_stop=False):
        out,writes,_=self.exercise_retire(late_stop=late_stop)
        base=Path('/synthetic');files={}
        for key,value in writes:
            target=base/'STOPPED.json' if key=='global_stop' else base/'retire'/(key+'.json')
            files[str(target)]=x.encoded(value)
        if malformed_stop:files[str(base/'STOPPED.json')]=x.encoded({'status':'STOPPED','secret':'SYNTHETIC'})
        def read(path,*_):
            raw=files[str(path)]
            if during_read and str(path)==str(base/'retire/result.json'):
                files[str(base/'STOPPED.json')]=x.encoded({'status':'STOPPED','binding':x.cleanup_binding(s),
                    'action':'retire','host':x.CP,'first_error':'HOST_DEADLINE','preserve':True,
                    'observed_at':x.iso(NOW),'retry':False})
            return raw
        with patch.object(x,'protected',side_effect=read),\
             patch.object(x.os.path,'lexists',side_effect=lambda path:str(path) in files):
            return x.collect_action(s,base,'retire',x.CP,NOW,(c,))
