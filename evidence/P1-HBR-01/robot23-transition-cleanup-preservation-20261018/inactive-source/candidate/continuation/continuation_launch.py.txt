def command_failure_detail(request,exc):
    """Public fallback when the relay STOP cannot be persisted. Never raw text."""
    from continuation_endpoint import public_command_detail
    source=getattr(exc,'remote_evidence',{})
    if not isinstance(source,dict):source={}
    source={k:source.get(k,getattr(exc,k,None if k=='command_audit' else [])) for k in ('command_evidence','command_audit')}
    rows=source['command_evidence'];report=source['command_audit'];scope=None
    if isinstance(report,dict):scope=report.get('scope')
    if scope is None and isinstance(rows,list) and rows and isinstance(rows[0],dict):
        match=re.fullmatch(r'command-(node-k8s-worker0[123]|worker01-file)-[0-9]{3}',str(rows[0].get('key','')))
        if match:scope=match.group(1)
    if scope is None:
        return {'command_evidence':[],'command_audit':None},([] if rows==[] and report is None else ['COMMAND_DETAIL_INVALID'])
    host=scope.removeprefix('node-') if isinstance(scope,str) and scope.startswith('node-') else 'k8s-worker01'
    return public_command_detail(request,{'key':scope,'host':host},source)

def protected(path):
    path=Path(path)
    for p in path.parents:
        s=p.lstat();check(stat.S_ISDIR(s.st_mode) and s.st_uid in (0,os.getuid()) and not s.st_mode&0o022)
    fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_CLOEXEC)
    try:
        s=os.fstat(fd);check(stat.S_ISREG(s.st_mode) and s.st_uid==os.getuid() and s.st_nlink==1 and
                            stat.S_IMODE(s.st_mode)==0o600 and s.st_size<=1048576)
        with os.fdopen(os.dup(fd),'rb') as f:raw=f.read(1048577)
        end=os.fstat(fd);check(len(raw)==s.st_size and all(getattr(s,k)==getattr(end,k) for k in
                     ('st_dev','st_ino','st_size','st_mtime_ns','st_ctime_ns')))
        return raw
    finally:os.close(fd)
